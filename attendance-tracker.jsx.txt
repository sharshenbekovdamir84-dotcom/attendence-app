import { useState, useEffect } from 'react';
import { ChevronLeft, ChevronRight, Plus, Trash2, Check, X, Loader2 } from 'lucide-react';

function toDateKey(d) {
  const yyyy = d.getFullYear();
  const mm = String(d.getMonth() + 1).padStart(2, '0');
  const dd = String(d.getDate()).padStart(2, '0');
  return `${yyyy}-${mm}-${dd}`;
}

function isSameDay(a, b) {
  return toDateKey(a) === toDateKey(b);
}

function formatDate(d) {
  const s = d.toLocaleDateString('ru-RU', { weekday: 'long', day: 'numeric', month: 'long' });
  return s.charAt(0).toUpperCase() + s.slice(1);
}

export default function AttendanceTracker() {
  const [students, setStudents] = useState([]);
  const [absentIds, setAbsentIds] = useState(new Set());
  const [currentDate, setCurrentDate] = useState(new Date());
  const [rosterLoaded, setRosterLoaded] = useState(false);
  const [dayLoaded, setDayLoaded] = useState(false);
  const [newName, setNewName] = useState('');
  const [errorMsg, setErrorMsg] = useState('');

  const dateKey = toDateKey(currentDate);
  const today = new Date();

  // Загружаем список учеников один раз при открытии
  useEffect(() => {
    (async () => {
      try {
        const res = await window.storage.get('roster', false);
        if (res && res.value) setStudents(JSON.parse(res.value));
      } catch (e) {
        // сохранённого списка ещё нет — начинаем с пустого
      } finally {
        setRosterLoaded(true);
      }
    })();
  }, []);

  // Загружаем отметки за выбранный день при каждой смене даты
  useEffect(() => {
    let cancelled = false;
    setDayLoaded(false);
    (async () => {
      try {
        const res = await window.storage.get(`attendance:${dateKey}`, false);
        if (cancelled) return;
        setAbsentIds(res && res.value ? new Set(JSON.parse(res.value)) : new Set());
      } catch (e) {
        if (!cancelled) setAbsentIds(new Set());
      } finally {
        if (!cancelled) setDayLoaded(true);
      }
    })();
    return () => {
      cancelled = true;
    };
  }, [dateKey]);

  async function persistRoster(list) {
    setStudents(list);
    try {
      await window.storage.set('roster', JSON.stringify(list), false);
    } catch (e) {
      setErrorMsg('Не удалось сохранить список — изменения могут пропасть после перезагрузки.');
    }
  }

  async function persistAttendance(set) {
    setAbsentIds(set);
    try {
      await window.storage.set(`attendance:${dateKey}`, JSON.stringify([...set]), false);
    } catch (e) {
      setErrorMsg('Не удалось сохранить отметку — проверьте соединение и попробуйте снова.');
    }
  }

  function toggleStudent(id) {
    const next = new Set(absentIds);
    next.has(id) ? next.delete(id) : next.add(id);
    persistAttendance(next);
  }

  function addStudent() {
    const name = newName.trim();
    if (!name) return;
    const id = `${Date.now().toString(36)}${Math.random().toString(36).slice(2, 6)}`;
    persistRoster([...students, { id, name }]);
    setNewName('');
  }

  function removeStudent(id) {
    persistRoster(students.filter((s) => s.id !== id));
    if (absentIds.has(id)) {
      const next = new Set(absentIds);
      next.delete(id);
      persistAttendance(next);
    }
  }

  function markAllPresent() {
    persistAttendance(new Set());
  }

  function changeDay(delta) {
    const d = new Date(currentDate);
    d.setDate(d.getDate() + delta);
    setCurrentDate(d);
  }

  const presentCount = students.length - absentIds.size;
  const loading = !rosterLoaded || !dayLoaded;
  const isToday = isSameDay(currentDate, today);

  return (
    <div className="min-h-screen bg-stone-200 flex justify-center px-3 py-6 sm:py-10">
      <div className="w-full max-w-md bg-stone-50 border border-stone-300">
        <div className="px-5 pt-5 pb-4 border-b border-stone-300">
          <h1 className="font-serif text-2xl text-stone-900">Журнал посещаемости</h1>

          <div className="flex items-center justify-between mt-3">
            <button
              onClick={() => changeDay(-1)}
              className="text-stone-400 hover:text-stone-700 transition-colors p-1 -ml-1"
              aria-label="Предыдущий день"
            >
              <ChevronLeft size={18} />
            </button>
            <div className="text-center">
              <div className="text-sm text-stone-600">{formatDate(currentDate)}</div>
              {!isToday && (
                <button
                  onClick={() => setCurrentDate(new Date())}
                  className="text-xs text-stone-400 underline decoration-stone-300 hover:text-stone-600 transition-colors"
                >
                  вернуться к сегодня
                </button>
              )}
            </div>
            <button
              onClick={() => changeDay(1)}
              className="text-stone-400 hover:text-stone-700 transition-colors p-1 -mr-1"
              aria-label="Следующий день"
            >
              <ChevronRight size={18} />
            </button>
          </div>

          <div className="flex gap-4 mt-3 font-mono text-sm">
            <span className="text-emerald-700">{presentCount} присутствуют</span>
            <span className="text-red-700">{absentIds.size} отсутствуют</span>
          </div>
        </div>

        {errorMsg && (
          <div className="px-5 py-2 text-sm text-red-700 bg-red-50 border-b border-stone-300">
            {errorMsg}
          </div>
        )}

        {loading ? (
          <div className="flex justify-center py-16 text-stone-300">
            <Loader2 className="animate-spin" size={22} />
          </div>
        ) : (
          <div>
            {students.length === 0 && (
              <p className="px-5 py-6 text-sm text-stone-400">
                Список пуст — добавьте первого ученика строкой ниже.
              </p>
            )}

            {students.map((s, i) => {
              const absent = absentIds.has(s.id);
              return (
                <div
                  key={s.id}
                  onClick={() => toggleStudent(s.id)}
                  className={`flex items-center gap-3 px-5 py-3 border-b border-stone-200 cursor-pointer ${
                    absent ? 'bg-red-50' : 'bg-transparent'
                  }`}
                >
                  <span className="font-mono text-sm text-stone-400 w-6 shrink-0">
                    {String(i + 1).padStart(2, '0')}
                  </span>
                  <span className="flex-1 text-stone-900 text-base">{s.name}</span>
                  {absent ? (
                    <X size={18} className="text-red-700 shrink-0" />
                  ) : (
                    <Check size={18} className="text-emerald-700 shrink-0" />
                  )}
                  <button
                    onClick={(e) => {
                      e.stopPropagation();
                      removeStudent(s.id);
                    }}
                    className="text-stone-300 hover:text-red-600 transition-colors p-1 shrink-0"
                    aria-label={`Удалить ${s.name}`}
                  >
                    <Trash2 size={15} />
                  </button>
                </div>
              );
            })}

            <div className="flex items-center gap-3 px-5 py-3 border-t border-dashed border-stone-300">
              <Plus size={16} className="text-stone-400 shrink-0" />
              <input
                type="text"
                value={newName}
                onChange={(e) => setNewName(e.target.value)}
                onKeyDown={(e) => e.key === 'Enter' && addStudent()}
                placeholder="Добавить ученика"
                className="flex-1 bg-transparent text-stone-900 placeholder-stone-400 text-base focus:outline-none"
              />
              {newName.trim() && (
                <button
                  onClick={addStudent}
                  className="text-sm text-stone-600 hover:text-stone-900 transition-colors"
                >
                  добавить
                </button>
              )}
            </div>
          </div>
        )}

        {students.length > 0 && (
          <div className="px-5 py-4 text-center">
            <button
              onClick={markAllPresent}
              className="text-xs text-stone-400 underline decoration-stone-300 hover:text-stone-600 transition-colors"
            >
              отметить всех присутствующими
            </button>
          </div>
        )}
      </div>
    </div>
  );
}
